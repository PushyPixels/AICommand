Create a 100 cube climbing course.  Each cube has a random size between 0.05f and 0.5f on each side, and each cube is randomly placed within 0.5f and 1.0f meter of the previous cube's position.  Every 5 cubes create a nearby light with a random color. */

using UnityEngine;
using UnityEditor;

public class ExampleEditor : Editor
{
	[MenuItem("Edit/Do Task")]
	static void DoTask()
	{
		float maxSideLength = 0.5f;
		float minSideLength = 0.05f;
		Vector3 prevPos = Vector3.zero;
		int lightCounter = 0;

		for (int i = 0; i < 100; i++)
		{
			var sideLength = Random.Range(minSideLength, maxSideLength);
			var cubePosition = new Vector3(prevPos.x + Random.Range(-0.5f, 1.0f), prevPos.y + Random.Range(-0.5f, 1.0f), prevPos.z + Random.Range(-0.5f, 1.0f));
			var cube = GameObject.CreatePrimitive(PrimitiveType.Cube);
			cube.transform.position = cubePosition;
			cube.GetComponent<BoxCollider>().size = new Vector3(sideLength, sideLength, sideLength);
			prevPos = cubePosition;

			if (i % 5 == 0 && lightCounter < 100)
			{
				GameObject lightObj = GameObject.CreatePrimitive(PrimitiveType.Sphere);
				lightObj.transform.position = cubePosition;
				Light lightComponent = lightObj.AddComponent<Light>();
				lightComponent.color = new Color(Random.Range(0f, 1f), Random.Range(0f, 1f), Random.Range(0f, 1f));
				lightComponent.range = 5f;
				lightCounter++;
			}
		}
	}
}
