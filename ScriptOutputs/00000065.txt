using UnityEngine;
using UnityEditor;

public class ClimbingCourseCreator : Editor
{
	[MenuItem("Edit/Create Climbing Course")]
	static void CreateClimbingCourse()
	{
		Vector3 lastPosition = new Vector3(0, 0, 0);
		float maxSize = 0.5f;
		float minSize = 0.05f;
		for (int i = 0; i < 100; i++)
		{
			Vector3 newPosition = lastPosition + (Vector3)Random.insideUnitSphere * Random.Range(0.5f, 1.0f);
			float cubeSize = minSize + Random.Range(0, maxSize - minSize);
			var cube = GameObject.CreatePrimitive(PrimitiveType.Cube);
			cube.transform.position = newPosition;
			cube.transform.localScale = new Vector3(cubeSize, cubeSize, cubeSize);
			lastPosition = newPosition;
		}
	}
}
