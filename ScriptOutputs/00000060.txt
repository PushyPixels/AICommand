using UnityEngine;
using UnityEditor;

public class ExampleEditor : Editor
{
    [MenuItem("Edit/Do Task")]
    static void DoTask()
    {
        for (int i = 0; i < 100; i++)
        {
            var light = new GameObject("Light");
            light.AddComponent<Light>();
            light.transform.position = new Vector3(Random.Range(-5f, 5f), Random.Range(-5f, 5f), Random.Range(-5f, 5f));
            
            // Assign random colors
            var color = new Color();
            color.r = Random.value;
            color.g = Random.value;
            color.b = Random.value;
            light.GetComponent<Light>().color = color;
        }
    }
}
