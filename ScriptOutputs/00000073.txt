using System;
using System.Collections.Generic;
using UnityEngine;
using UnityEditor;

public class EncouragingMessages : EditorWindow
{
	[MenuItem("Edit/Do Task")]
	static void DoTask()
	{
		var textMeshes = FindObjectsOfType<TextMesh>();
		List<string> messages = new List<string>() { "Keep going!", "Almost there.", "You've got this.", "Persevere.", "Stay focused." };
		Random random = new Random();

		for (int i = 0; i < textMeshes.Length; i++)
		{
			TextMesh mesh = textMeshes[i];
			mesh.text = messages[random.Next(messages.Count)];
		}
	}
}
