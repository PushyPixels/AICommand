using UnityEngine;
using UnityEditor;

public class ExampleEditor : Editor
{
	[MenuItem("Edit/Do Task")]
	static void DoTask()
	{
		Vector3 previousPosition = Vector3.zero;
		for (int i = 0; i < 100; i++)
		{
			var cubeSideLength = Random.Range(0.05f, 0.5f);
			var currentPosition = new Vector3(
				previousPosition.x + Random.Range(-0.5f, 1.0f),
				previousPosition.y + Random.Range(-0.5f, 1.0f),
				previousPosition.z + Random.Range(-0.5f, 1.0f)
			);
			var cube = GameObject.CreatePrimitive(PrimitiveType.Cube);
			cube.transform.localScale = new Vector3(cubeSideLength, cubeSideLength, cubeSideLength);
			cube.transform.position = currentPosition;
			previousPosition = currentPosition;
		}
	}
}
