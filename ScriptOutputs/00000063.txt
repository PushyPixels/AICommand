using UnityEngine;
using UnityEditor;

public class ClimbingCourseGenerator : Editor
{
	[MenuItem("Edit/Do Task")]
	static void DoTask()
	{
		for (int i = 0; i < 100; i++)
		{
			float sizeRange = Random.Range(2f, 4f);
			var cubeSize = new Vector3(sizeRange, sizeRange, sizeRange);
			Vector3 currentPosition;

			if (i > 0)
			{
				// Find the last created game object
				GameObject lastCube = GameObject.Find("Cube" + (i - 1));
				currentPosition = lastCube.transform.position;
			}
			else
			{
				currentPosition = new Vector3(Random.Range(-5f, 5f), Random.Range(-5f, 5f), Random.Range(-5f, 5f));
			}

			Vector3 deltaPosition = Vector3.zero;
			if (Mathf.Abs(currentPosition.x) < 1)
			{
				deltaPosition.x = Random.Range(-0.5f, 0.5f);
			}
			if (Mathf.Abs(currentPosition.y) < 1)
			{
				deltaPosition.y = Random.Range(-0.5f, 0.5f);
			}
			if (Mathf.Abs(currentPosition.z) < 1)
			{
				deltaPosition.z = Random.Range(-0.5f, 0.5f);
			}
			currentPosition += deltaPosition;

			var cube = GameObject.CreatePrimitive(PrimitiveType.Cube);
			cube.transform.localScale = cubeSize;
			cube.name = "Cube" + i;
			cube.transform.position = currentPosition;
		}
	}
}
