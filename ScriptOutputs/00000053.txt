/* Write a Unity Editor script.
 - It provides its functionality as a menu item placed "Edit" > "Do Task".
 - It doesn’t provide any editor window. It immediately does the task when the menu item is invoked.
 - Don’t use GameObject.FindGameObjectsWithTag.
 - There is no selected object. Find game objects manually.
 - I only need the script body. Don’t add any explanation.
 - Don't forget to add "using UnityEditor;"
The task is described as follows:
Create 100 cubes at random points. */
```csharp
using UnityEngine;
using System.Collections.Generic;
using Random = UnityEngine.Random;

[InitializeOnLoad]
public class CreateCubes : UnityEditor.MenuItemCallback
{
    static CreateCubes() {
        var path = "Edit/Do Task";
        UnityEditor.Menu.AddCommand(new GUIContent("Create 100 Cubes"), false, DoTask, path);
    }
    
    public static void DoTask()
    {
        List<Transform> cubes = new List<Transform>();
        for (int i = 0; i < 100; ++i)
        {
            Vector3 position = Random.insideUnitSphere * 10f + Vector3.one * 5f;
            GameObject cube = Object.Instantiate(Resources.Load("Cube")) as GameObject;
            cube.transform.position = position;
            cube.transform.parent = transform;
            cubes.Add(cube.transform);
        }
    }
}
```
