using UnityEngine;
using UnityEditor;

public class HumanoidCubeCreator : Editor
{
	[MenuItem("Edit/Do Task")]
	static void DoTask()
	{
		var torsoPosition = new Vector3(0f, 1.5f, 0f);
		GameObject torso = GameObject.CreatePrimitive(PrimitiveType.Cube);
		torso.transform.position = torsoPosition;

		var headPosition = torsoPosition;
		headPosition.y += 2f;
		GameObject head = GameObject.CreatePrimitive(PrimitiveType.Cube);
		head.transform.parent = torso.transform;
		head.transform.localPosition = Vector3.up * head.GetComponent<BoxCollider>().size.y / 2;
		head.transform.position += headPosition;

		var leftArmPosition = torsoPosition;
		leftArmPosition.x -= 2f;
		GameObject leftArm = GameObject.CreatePrimitive(PrimitiveType.Cube);
		leftArm.transform.parent = torso.transform;
		leftArm.transform.localPosition = Vector3.forward * leftArm.GetComponent<BoxCollider>().size.x / 2;
		leftArm.transform.position += leftArmPosition;

		var rightArmPosition = torsoPosition;
		rightArmPosition.x += 2f;
		GameObject rightArm = GameObject.CreatePrimitive(PrimitiveType.Cube);
		rightArm.transform.parent = torso.transform;
		rightArm.transform.localPosition = -Vector3.forward * rightArm.GetComponent<BoxCollider>().size.x / 2;
		rightArm.transform.position += rightArmPosition;

		var leftLegPosition = torsoPosition;
		leftLegPosition.y -= 5f;
		GameObject leftLeg = GameObject.CreatePrimitive(PrimitiveType.Cube);
		leftLeg.transform.parent = torso.transform;
		leftLeg.transform.localRotation = Quaternion.Euler(0f, 180f, 0f);
		leftLeg.transform.localPosition = Vector3.back * leftLeg.GetComponent<BoxCollider>().size.z / 2;
		leftLeg.transform.position += leftLegPosition;

		var rightLegPosition = torsoPosition;
		rightLegPosition.x += 2f;
		rightLegPosition.y -= 5f;
		GameObject rightLeg = GameObject.CreatePrimitive(PrimitiveType.Cube);
		rightLeg.transform.parent = torso.transform;
		rightLeg.transform.localRotation = Quaternion.Euler(0f, 180f, 90f);
		rightLeg.transform.localPosition = Vector3.back * rightLeg.GetComponent<BoxCollider>().size.z / 2;
		rightLeg.transform.position += rightLegPosition;
	}
}
