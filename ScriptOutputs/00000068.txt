using UnityEngine;
using UnityEditor;

public class ExampleEditor : Editor
{
	[MenuItem("Edit/Do Task")]
	static void DoTask()
	{
		for (int i = 0; i < 100; i++)
		{
			var sizeRange = new Vector3(0.05f, 0.05f, 0.05f);
			sizeRange.x *= Random.value + 9;
			sizeRange.y *= Random.value + 9;
			sizeRange.z *= Random.value + 9;

			var cubePosition = new Vector3(Random.Range(-5f, 5f), Random.Range(-5f, 5f), Random.Range(-5f, 5f));
			if (i > 0)
			{
				cubePosition.x = Mathf.Clamp(cubePosition.x, previousCubePosition.x - 0.5f, previousCubePosition.x + 0.5f);
				cubePosition.y = Mathf.Clamp(cubePosition.y, previousCubePosition.y - 0.5f, previousCubePosition.y + 0.5f);
				cubePosition.z = Mathf.Clamp(cubePosition.z, previousCubePosition.z - 0.5f, previousCubePosition.z + 0.5f);
			}

			GameObject cube = GameObject.CreatePrimitive(PrimitiveType.Cube);
			cube.transform.localScale = sizeRange;
			cube.transform.position = cubePosition;
			previousCubePosition = cubePosition;

			if (i % 5 == 0)
			{
				GameObject nearbyLight = new GameObject();
				nearbyLight.AddComponent<Light>();
				nearbyLight.GetComponent<Light>().color = new Color(Random.Range(0f, 1f), Random.Range(0f, 1f), Random.Range(0f, 1f));
				nearbyLight.transform.position = cubePosition;
			}
		}
	}

	static Vector3 previousCubePosition = new Vector3(-5f, -5f, -5f);
}
