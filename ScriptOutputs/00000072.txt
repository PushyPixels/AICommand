```csharp
using UnityEngine;
using UnityEditor;

public class EncourageTexts : EditorWindow
{
	[MenuItem("Edit/Do Task")]
	static void DoTask()
	{
		var textMeshes = FindObjectsOfType<TextMesh>();
		string[] messages = new string[] { "You've got this!", "Keep pushing forward!", "Don't give up, you're almost there!" };

		for (int i = 0; i < textMeshes.Length; i++)
		{
			textMeshes[i].text = messages[i];
		}
	}
}
```
This script will iterate through all TextMesh objects in the scene and assign a random encouraging message from the provided array. Each text mesh should display a unique message as `messages[i]` is used for each iteration of the loop.
