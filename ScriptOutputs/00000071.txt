using UnityEngine;
using UnityEditor;

public class ExampleEditor : Editor
{
	[MenuItem("Edit/Do Task")]
	static void DoTask()
	{
		for (int i = 0; i < 10; i++)
		{
			var go = new GameObject("Warning" + i);
			go.transform.position = new Vector3(Random.Range(-5f, 5f), Random.Range(-5f, 5f), Random.Range(-5f, 5f));

			// Remove existing Mesh Renderer
			MeshRenderer[] meshRenderers = go.GetComponentsInChildren<MeshRenderer>(true);
			foreach (var mr in meshRenderers)
			{
				DestroyImmediate(mr.gameObject);
			}

			// Create and set Text Mesh component
			TextMesh warning = go.AddComponent<TextMesh>();
			warning.text = "Warning: " + GetCreativeWarning();
			warning.anchor = TextAnchor.MiddleCenter;
		}
	}

	static string GetCreativeWarning()
	{
		List<string> warnings = new List<string> { 
			"This is not a drill, avoid this area.",
			"Danger: Do not proceed without proper gear.",
			"Caution: Area is unstable.",
			"Unauthorized access will be punished by lasers.",
		};
		return warnings[Random.Range(0, warnings.Count)];
	}
}
