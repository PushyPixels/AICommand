using UnityEngine;
using UnityEditor;

public class ExampleEditor : Editor
{
	[MenuItem("Edit/Do Task")]
	static void DoTask()
	{
		Transform current = null;
		for (int i = 0; i < 100; i++)
		{
			var size = Random.Range(0.1f, 1.0f);
			GameObject cube = GameObject.CreatePrimitive(PrimitiveType.Cube);
			cube.transform.localScale = new Vector3(size, size, size);
			Vector3 pos;
			if (i > 0)
			{
				pos = current.position + Random.insideUnitSphere * Random.Range(0f, 1f);
			}
			else
			{
				pos = Vector3.zero;
			}
			cube.transform.position = pos;
			current = cube.transform;
		}
	}
}
