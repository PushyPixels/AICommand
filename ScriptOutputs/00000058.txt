 ```csharp
 using System;
 using UnityEngine;

 public class CreateLights : UnityEditor.MenuCallbacks
 {
     public static void CreateRandomLights()
     {
         for (int i = 0; i < 100; ++i)
         {
             var lightPos = new Vector3(Random.Range(-10f, 10f), Random.Range(-5f, 5f), Random.Range(-10f, 10f));
             GameObject newLight = new GameObject("Light" + i);
             newLight.transform.position = lightPos;

             var color = new Color32(Random.Range(0, 256), Random.Range(0, 256), Random.Range(0, 256), 1);
             var light = new Light {color = color};
             light.type = LightType.Point;
             newLight.AddComponent<Light>();
             newLight.GetComponent<Light>().pointLight = light as PointLight;
         }
     }

     static CreateLights()
     {
         UnityEditor.Menu.AddMenuItem(new Vector2(10, 3), false, 54, "Do Task/Create 100 Random Lights");
     }
 }
```
