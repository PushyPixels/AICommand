using UnityEngine;
using UnityEditor;

public class ExampleEditor : Editor
{
    [MenuItem("Edit/Do Task")]
    static void DoTask()
    {
        var cubes = FindObjectsOfType<GameObject>();
        foreach (var cube in cubes)
        {
            if (cube.name == "Cube")
            {
                cube.transform.rotation = Quaternion.Euler(new Vector3(Random.Range(-360f, 360f), Random.Range(-360f, 360f), Random.Range(-360f, 360f)));
            }
        }
    }
}
