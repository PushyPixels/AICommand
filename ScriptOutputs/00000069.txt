using UnityEngine;
using UnityEditor;

public class ExampleEditor : Editor
{
	[MenuItem("Edit/Do Task")]
	static void DoTask()
	{
		string[] warnings = { "Warning 1", "Warning 2", "Warning 3" };
		for (int i = 0; i < 5; i++)
		{
			var textMesh = GameObject.CreatePrimitive(PrimitiveType.Quad);
			textMesh.AddComponent<TextMesh>();
			textMesh.name = "WarningText";
			textMesh.transform.position = new Vector3(Random.Range(-5f, 5f), Random.Range(-5f, 5f), Random.Range(-5f, 5f));
			textMesh.GetComponent<TextMesh>().text = warnings[i % warnings.Length];
		}
	}
}
