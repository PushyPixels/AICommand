using UnityEngine;
using UnityEditor;

public class ExampleEditor : Editor
{
    [MenuItem("Edit/Do Task")]
    static void DoTask()
    {
        for (int i = 0; i < 100; i++)
        {
            var light = new GameObject("Light");
            light.AddComponent<PointLight>();
            
            var pointLight = light.GetComponent<PointLight>();
            pointLight.range = 50f;
            pointLight.intensity = Random.Range(1, 8);
            pointLight.color = new Color(Random.Range(0, 1), Random.Range(0, 1), Random.Range(0, 1));
            
            light.transform.position = new Vector3(Random.Range(-5f, 5f), Random.Range(-5f, 5f), Random.Range(-5f, 5f));
        }
    }
}
